"""In-memory key-value store with an audit log."""


class Store:
    def __init__(self):
        self._data = {}
        self.audit = []

    def set(self, key, value):
        self._data[key] = value
        self.audit.append(("set", key))

    def get(self, key, default=None):
        return self._data.get(key, default)

    def rename(self, old, new):
        if old not in self._data:
            return False
        if isinstance(new, str) and new.startswith("sys:"):
            raise ValueError("invalid target")
        if new != old:
            if new in self._data:
                raise ValueError("target exists")
            self._data[new] = self._data.pop(old)
        self.audit.append(("rename", old, new))
        return True

    def delete(self, key):
        if key in self._data:
            del self._data[key]
            self.audit.append(("delete", key))
            return True
        return False

import unittest

from store import Store


class TestStore(unittest.TestCase):
    def test_set_get_roundtrip(self):
        s = Store()
        s.set("a", 1)
        self.assertEqual(s.get("a"), 1)

    def test_get_default(self):
        s = Store()
        self.assertIsNone(s.get("missing"))
        self.assertEqual(s.get("missing", 0), 0)

    def test_rename(self):
        s = Store()
        s.set("a", 1)
        self.assertTrue(s.rename("a", "b"))
        self.assertIsNone(s.get("a"))
        self.assertEqual(s.get("b"), 1)

    def test_rename_failure_paths(self):
        s = Store()
        s.set("a", 1)
        s.set("b", 2)
        self.assertFalse(s.rename("missing", "c"))
        with self.assertRaises(ValueError):
            s.rename("a", "sys:x")
        with self.assertRaises(ValueError):
            s.rename("a", "b")
        self.assertEqual(s.get("a"), 1)
        self.assertEqual(s.get("b"), 2)
        self.assertNotIn(("rename", "a", "b"), s.audit)

    def test_delete(self):
        s = Store()
        s.set("a", 1)
        self.assertTrue(s.delete("a"))
        self.assertIsNone(s.get("a"))
        self.assertFalse(s.delete("a"))


if __name__ == "__main__":
    unittest.main(verbosity=2)
